🔧 How to Fix This:

Add server-side authorization checks:

@app.route('/admin')
def admin_panel():
    # Check if user is logged in
    if 'username' not in session:
        abort(401)  # Unauthorized
    
    # Check if user has admin role
    if session.get('role') != 'admin':
        abort(403)  # Forbidden
    
    return render_template('admin.html')
